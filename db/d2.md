# Реляционная модель

```sql
CREATE TABLE person (
    id          VARCHAR(50)     PRIMARY KEY,
    first_name  VARCHAR(250),
    last_name   VARCHAR(250),
    gender      CHAR(1)         CHECK (gender IN ('M', 'F', 'U')),
    spouse_id   VARCHAR(50)     REFERENCES person(id),
    UNIQUE(spouse_id),
    CHECK (id != spouse_id)
);

CREATE TABLE parent_link (
    child_id    VARCHAR(50) NOT NULL REFERENCES person(id),
    parent_id   VARCHAR(50) NOT NULL REFERENCES person(id),
    parent_role CHAR(1)     NOT NULL CHECK (parent_role IN ('F', 'M')),
    PRIMARY KEY (child_id, parent_id),
    CHECK (child_id != parent_id)
);

CREATE TABLE sibling_link (
    person_id    VARCHAR(50) NOT NULL REFERENCES person(id),
    sibling_id   VARCHAR(50) NOT NULL REFERENCES person(id),
    sibling_type CHAR(1)     NOT NULL CHECK (sibling_type IN ('B', 'S')),
    PRIMARY KEY (person_id, sibling_id),
    CHECK (person_id != sibling_id)
);

CREATE OR REPLACE FUNCTION check_parent_gender()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.parent_role = 'F' THEN
        IF (SELECT gender FROM person WHERE id = NEW.parent_id) NOT IN ('M', 'U') THEN
            RAISE EXCEPTION 'Отец должен быть мужского пола';
        END IF;
    ELSIF NEW.parent_role = 'M' THEN
        IF (SELECT gender FROM person WHERE id = NEW.parent_id) NOT IN ('F', 'U') THEN
            RAISE EXCEPTION 'Мать должна быть женского пола';
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_check_parent_gender
BEFORE INSERT OR UPDATE ON parent_link
FOR EACH ROW EXECUTE FUNCTION check_parent_gender();

CREATE OR REPLACE FUNCTION check_spouse_gender()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.spouse_id IS NOT NULL AND NEW.gender IS NOT NULL THEN
        IF (SELECT gender FROM person WHERE id = NEW.spouse_id) = NEW.gender THEN
            RAISE EXCEPTION 'Супруги должны быть разного пола';
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_check_spouse_gender
BEFORE INSERT OR UPDATE ON person
FOR EACH ROW EXECUTE FUNCTION check_spouse_gender();

CREATE VIEW sibling_view AS
SELECT
    p1.id AS person_id,
    p2.id AS sibling_id
FROM person p1
JOIN person p2 ON (
    EXISTS (
        SELECT 1 FROM parent_link pl1
        JOIN parent_link pl2 ON pl1.parent_id = pl2.parent_id
        WHERE pl1.child_id = p1.id AND pl2.child_id = p2.id
    )
)
WHERE p1.id != p2.id;
```
